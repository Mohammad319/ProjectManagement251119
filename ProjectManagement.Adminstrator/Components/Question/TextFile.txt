@code {
    [Parameter] public int TaskId { get; set; }

    // ===== State =====
    private bool _loading = true;
    private bool _saving = false;
    private string? _error;

    // ===== Data =====
    private TaskEXMModel? _task;
    private List<QuestionEquation> _qFormulas = new();

    // لكل Choice (qeId + choice.Label)
    private Dictionary<string, ChoiceBindingVM> _choiceMap = new(StringComparer.Ordinal);

    // تجهيز صفوف موارد الشروط مرة واحدة: ConditionId -> rows
    private sealed class CondRowDto
    {
        public int AssignmentId { get; init; }
        public int ResourceId { get; init; }
        public string? ResourceName { get; init; }
    }
    private Dictionary<int, List<CondRowDto>> _condRowsByCond = new();

    // طيّ افتراضيًا
    private HashSet<int> _expandedQEs = new();
    private HashSet<int> _expandedConditions = new();
    private const int PreviewCount = 10;

    protected override async Task OnParametersSetAsync() => await LoadAsync();

    private async Task LoadAsync()
    {
        _loading = true; _error = null;
        try
        {
            await using var db = await ContextFactory.CreateDbContextAsync();

            // 1) المهمة + مواردها
            _task = await db.Tasks
                            .AsNoTracking()
                            .Select(t => new TaskEXMModel
                            {
                                Id = t.Id,
                                Name = t.Name,
                                Code = t.Code,
                                TaskResources = t.TaskResources.Select(ts => new TaskResourceModel
                                {
                                    Id = ts.Id,
                                    ResourceId = ts.ResourceId,
                                    Resource = ts.Resource == null ? null : new ResourceEXModel
                                    {
                                        Name = ts.Resource.Name
                                    }
                                }).ToList()
                            }).FirstOrDefaultAsync(t => t.Id == TaskId);

            if (_task is null) { _error = $"لم يتم العثور على المهمة {TaskId}."; _loading = false; return; }

            // 2) أسئلة المهمة (بدون Choices) + شروط + ResourceAssignments (+ Resource للأسماء)
            _task.Questions = await db.Questions
              .AsNoTracking()
              .Where(q => q.TaskId == TaskId)
              .Include(q => q.Conditions)
                .ThenInclude(c => c.ResourceAssignments)
                  .ThenInclude(ra => ra.Resource)
              .Include(q => q.Conditions)
                .ThenInclude(c => c.ResourceChoice)
                  .ThenInclude(rc => rc.Resource)
              .Include(q => q.Conditions)
                .ThenInclude(c => c.Choice)
              .OrderBy(q => q.Id)
              .ToListAsync();

            // 3) QuestionFormulas + Choices (مصدر الأعمدة)
            _qFormulas = await db.QuestionFormulas
              .AsNoTracking()
              .Include(qe => qe.Choices)
              .Where(qe => qe.TaskId == TaskId)
              .OrderBy(qe => qe.Id)
              .ToListAsync();

            // 4) بناء صفوف موارد الشروط مرة واحدة
            _condRowsByCond.Clear();
            foreach (var q in _task!.Questions ?? Enumerable.Empty<Question>())
            {
                foreach (var c in q.Conditions ?? Enumerable.Empty<QuestionCondition>())
                {
                    if ((c.ResourceAssignments?.Count ?? 0) == 0) continue;

                    var rows = c.ResourceAssignments!
                      .OrderBy(a => a.ResourceId)
                      .Select(a => new CondRowDto
                      {
                          AssignmentId = a.Id,
                          ResourceId = a.ResourceId,
                          ResourceName = a.Resource?.Name
                      })
                      .ToList();

                    _condRowsByCond[c.Id] = rows;
                }
            }

            // 5) بناء حالة الاختيارات من التخزين الحالي
            _choiceMap.Clear();
            foreach (var qe in _qFormulas)
            {
                foreach (var ch in qe.Choices ?? Enumerable.Empty<ChoiceEquation>())
                {
                    var key = ChoiceKey(qe, ch);
                    var bind = ch.Resources?.FirstOrDefault();
                    _choiceMap[key] = new ChoiceBindingVM
                    {
                        TaskResourceIds = (bind?.TaskResourceIds ?? new()).ToHashSet(),
                        ResourceAssignmentIds = (bind?.ResourceAssignmentIds ?? new()).ToHashSet()
                    };
                }
            }

            // تأكد من وجود VM لكل Choice
            foreach (var qe in _qFormulas)
                foreach (var ch in qe.Choices ?? Enumerable.Empty<ChoiceEquation>())
                    EnsureVM(ChoiceKey(qe, ch));
        }
        catch (Exception ex) { _error = ex.Message; }
        finally { _loading = false; }
    }

    // === Expand/collapse ===
    private void ToggleQeExpand(int qeId)
    {
        if (!_expandedQEs.Add(qeId))
            _expandedQEs.Remove(qeId);
    }

    private void ToggleConditionExpand(int conditionId)
    {
        if (!_expandedConditions.Add(conditionId))
            _expandedConditions.Remove(conditionId);
    }

    // === Toggles ===
    private void ToggleTaskResource(string key, int resourceId, bool value)
    {
        if (!_choiceMap.TryGetValue(key, out var vm)) return;
        if (value) vm.TaskResourceIds.Add(resourceId);
        else vm.TaskResourceIds.Remove(resourceId);
    }

    private void ToggleAssignmentResource(string key, int assignmentId, bool value)
    {
        if (!_choiceMap.TryGetValue(key, out var vm)) return;
        if (value) vm.ResourceAssignmentIds.Add(assignmentId);
        else vm.ResourceAssignmentIds.Remove(assignmentId);
    }

    // === Save per QuestionEquation ===
    private async Task SaveForEquationAsync(QuestionEquation qe)
    {
        _saving = true; _error = null;
        try
        {
            await using var db = await ContextFactory.CreateDbContextAsync();

            var entity = await db.QuestionFormulas
                                 .Include(x => x.Choices)
                                 .FirstOrDefaultAsync(x => x.Id == qe.Id && x.TaskId == TaskId);

            if (entity is null) { _error = $"لم يتم العثور على QuestionEquation #{qe.Id}."; return; }

            var byLabel = entity.Choices.ToDictionary(c => c.Label ?? "", c => c, StringComparer.Ordinal);

            foreach (var ch in qe.Choices ?? Enumerable.Empty<ChoiceEquation>())
            {
                var label = ch.Label ?? "";
                if (!byLabel.TryGetValue(label, out var ce)) continue;

                var key = ChoiceKey(qe, ch);
                if (!_choiceMap.TryGetValue(key, out var vm)) continue;

                ce.Resources ??= new();
                var bind = ce.Resources.FirstOrDefault();
                if (bind is null)
                {
                    bind = new ChoiceEquationBinding();
                    ce.Resources.Add(bind);
                }

                bind.TaskResourceIds = vm.TaskResourceIds.OrderBy(i => i).ToList();
                bind.ResourceAssignmentIds = vm.ResourceAssignmentIds.OrderBy(i => i).ToList();
            }

            await db.SaveChangesAsync();
            _error = "تم الحفظ بنجاح.";
        }
        catch (Exception ex) { _error = ex.Message; }
        finally { _saving = false; }
    }

    // === Helpers ===
    private static string ChoiceKey(QuestionEquation qe, ChoiceEquation ch) => $"{qe.Id}::{ch.Label}";

    private void EnsureVM(string key)
    {
        if (!_choiceMap.ContainsKey(key))
            _choiceMap[key] = new ChoiceBindingVM();
    }

    private class ChoiceBindingVM
    {
        public HashSet<int> TaskResourceIds { get; set; } = new();
        public HashSet<int> ResourceAssignmentIds { get; set; } = new(); // store ResourceAssignment.Id
    }
}
